> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the Programme Manager one page showing every partnership lead's monthly MBO commitments grouped by urgency, fed by uploading the leads' existing Excel files.

**Architecture:** A new `DB.teamMbo` array collection, a `Team MBO` page replacing the `Team OKR` nav item, and an importer that determines the shape of each month block in the uploaded sheet independently (block heading → its own header row → its own column map) because real files change column order between months. A Dashboard card surfaces overdue items, replacing the `Documents due soon` widget.

**Tech Stack:** Single-file vanilla JS/HTML/CSS, no build step, no framework. `xlsx.full.min.js` (already loaded from CDN at line 15) reads the workbook. Firebase Realtime Database for persistence. Tests are plain Node scripts using `assert`.

**Design spec:** `docs/superpowers/specs/2026-08-18-team-mbo-design.md`

```js
/* Pull named functions out of the app's index.html and make them callable,
   so tests exercise the shipped source instead of a hand-copied duplicate.
   Extraction is by brace matching from `function NAME(` to its closing brace. */
const fs = require('fs');
const path = require('path');
const vm = require('vm');

const APP = path.join(__dirname, '..', 'index.html');

function extract(src, name){
  const start = src.search(new RegExp('function\\s+' + name + '\\s*\\('));
  if(start === -1) return null;
  const open = src.indexOf('{', start);
  if(open === -1) return null;
  let depth = 0, inStr = null, prev = '';
  for(let i = open; i < src.length; i++){
    const c = src[i];
    if(inStr){
      if(c === inStr && prev !== '\\') inStr = null;
    } else if(c === '"' || c === "'" || c === '`'){
      inStr = c;
    } else if(c === '{'){
      depth++;
    } else if(c === '}'){
      depth--;
      if(depth === 0) return src.slice(start, i + 1);
    }
    prev = c;
  }
  return null;
}

function loadFns(names, globals){
  const src = fs.readFileSync(APP, 'utf8');
  const missing = [], bodies = [];
  for(const n of names){
    const body = extract(src, n);
    if(!body) missing.push(n); else bodies.push(body);
  }
  if(missing.length) throw new Error('index.html has no function(s): ' + missing.join(', '));

  const sandbox = Object.assign({ console }, globals);
  vm.createContext(sandbox);
  vm.runInContext(bodies.join('\n\n'), sandbox);

  const out = {};
  for(const n of names) out[n] = sandbox[n];
  return out;
}

module.exports = { loadFns, extract };
```

```js
/* =====================================================================
   TEAM MBO · IMPORT PARSING
   The leads' spreadsheets stack one month block per section, and the
   column order changes between blocks in the same file. Everything here
   is pure so it can be tested without a browser.
   ===================================================================== */

const TM_MONTHS = ["january","february","march","april","may","june",
                   "july","august","september","october","november","december"];

function tmNormStatus(raw){
  const s = String(raw==null?"":raw).toLowerCase().replace(/\s+/g,"");
  if(!s) return "";
  if(s.startsWith("done") || s.startsWith("complete")) return "Done";
  if(s.includes("progress")) return "In Progress";
  /* "note done" is a real typo for "not done" in a live file */
  if(s.startsWith("not") || s.startsWith("note")) return "Not Done";
  return "";
}

/* Excel day 1 is 1900-01-01, with the well-known 1900 leap-year fudge,
   which makes day 0 equal 1899-12-30. */
function tmSerialToIso(n){
  const ms = Date.UTC(1899,11,30) + Math.round(n) * 86400000;
  return new Date(ms).toISOString().slice(0,10);
}

function tmIsoOf(y,m,d){
  if(m<1 || m>12 || d<1 || d>31) return "";
  const dt = new Date(Date.UTC(y,m-1,d));
  /* rejects 31 April and friends — the Date would roll into the next month */
  if(dt.getUTCMonth() !== m-1 || dt.getUTCDate() !== d) return "";
  return dt.toISOString().slice(0,10);
}

function tmParseDate(cell){
  if(cell==null) return "";
  if(typeof cell === "number" && cell > 20000 && cell < 80000) return tmSerialToIso(cell);
  const s = String(cell).trim();
  if(!s || s === "-") return "";

  /* a serial that arrived as text, e.g. "45884.0" */
  if(/^\d{4,5}(\.0+)?$/.test(s)){
    const n = parseFloat(s);
    if(n > 20000 && n < 80000) return tmSerialToIso(n);
  }
  /* d/m/yyyy or d-m-yyyy — day first */
  let m = /^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})$/.exec(s);
  if(m) return tmIsoOf(+m[3], +m[2], +m[1]);

  /* "28th Feb 2026", "31th July 2026", "30 June 2026" */
  m = /^(\d{1,2})\s*(?:st|nd|rd|th)?\s+([A-Za-z]+),?\s+(\d{4})$/.exec(s);
  if(m){
    const mi = TM_MONTHS.findIndex(x => x.startsWith(m[2].toLowerCase().slice(0,3)));
    if(mi >= 0) return tmIsoOf(+m[3], mi+1, +m[1]);
  }
  /* "August 15 2026" */
  m = /^([A-Za-z]+)\s+(\d{1,2})(?:st|nd|rd|th)?,?\s+(\d{4})$/.exec(s);
  if(m){
    const mi = TM_MONTHS.findIndex(x => x.startsWith(m[1].toLowerCase().slice(0,3)));
    if(mi >= 0) return tmIsoOf(+m[3], mi+1, +m[2]);
  }
  return "";
}

/* Some deadlines are already wrong in the source file: a lead typed
   "10/3/2026" meaning 10 March and Excel stored 3 October. Swap back only
   when the evidence is unambiguous — the day equals the block's month
   number, and swapping lands the date inside the block's month. Validated
   against every such date in two real files with no counter-examples. */
function tmRepairDate(iso, blockMonth){
  if(!iso || !blockMonth) return { iso: iso||"", fixed:false };
  const y = +iso.slice(0,4), m = +iso.slice(5,7), d = +iso.slice(8,10);
  if(m === blockMonth) return { iso, fixed:false };
  if(d !== blockMonth)  return { iso, fixed:false };
  const swapped = tmIsoOf(y, d, m);
  if(!swapped) return { iso, fixed:false };
  return { iso: swapped, fixed:true };
}

const TM_COLS = {
  objective: ["objective","goal"],
  title:     ["keyresult","actionitem"],
  deadline:  ["deadline","timeline","when","targetdate","duedate"],
  status:    ["status"],
  priority:  ["priority"],
  timeEst:   ["timeestimate"],
  timeSpent: ["timespent"],
  comment:   ["comment","notes","remark"],
  /* "resourse" is a real misspelling in a live file */
  ref:       ["reference","resource","resourse"],
  owner:     ["whom","bywhom"],
  objNum:    ["srno","srnumber","no"]
};

function tmColMap(cells){
  const map = {};
  (cells||[]).forEach((raw, i) => {
    const norm = String(raw==null?"":raw).toLowerCase().replace(/[^a-z]/g,"");
    if(!norm) return;
    for(const field of Object.keys(TM_COLS)){
      if(map[field] !== undefined) continue;
      /* longest-first so "timeestimate" never matches the "time" in another
         field, and exact match beats substring */
      if(TM_COLS[field].some(n => norm === n)){ map[field] = i; return; }
    }
  });
  /* second pass: substring matches for headers with extra words, e.g.
     "July Action Item" or "Target Date (final)" */
  (cells||[]).forEach((raw, i) => {
    const norm = String(raw==null?"":raw).toLowerCase().replace(/[^a-z]/g,"");
    if(!norm) return;
    for(const field of Object.keys(TM_COLS)){
      if(map[field] !== undefined) continue;
      if(TM_COLS[field].some(n => norm.includes(n))){ map[field] = i; return; }
    }
  });
  return map;
}

function tmRowCells(row){ return Array.isArray(row) ? row : []; }
function tmNonEmpty(row){
  return tmRowCells(row).filter(c => c!=null && String(c).trim() !== "");
}

/* Find each month block: its heading, its own header row, and its row range.
   kind is "kr" (key results) or "action" (action items). */
function tmFindBlocks(rows, kind){
  const headerNeedle = kind === "action" ? /action\s*item/i : /key\s*result/i;
  const blocks = [];

  for(let i = 0; i < rows.length; i++){
    const cells = tmNonEmpty(rows[i]);
    /* A heading row carries exactly one value. This is what rejects prose
       such as "...will send MBO by 5 January 2026" sitting in the Comment
       column of a fully populated data row. */
    if(cells.length !== 1) continue;

    const text = String(cells[0]).trim();
    const mi = TM_MONTHS.findIndex(mn => new RegExp("\\b" + mn + "\\b", "i").test(text));
    if(mi < 0) continue;
    const yearM = /\b(20\d{2})\b/.exec(text);

    if(kind === "kr"){
      /* key-result headings always name the sheet, e.g. "MBO - August 2026"
         or "MBO_Someone_August 2026" */
      if(!/\bMBO\b/i.test(text)) continue;
      if(!yearM) continue;
    }

    /* the header row must appear within the next 3 rows, else this was prose */
    let headerRow = -1;
    for(let j = i + 1; j <= Math.min(i + 3, rows.length - 1); j++){
      if(tmRowCells(rows[j]).some(c => headerNeedle.test(String(c==null?"":c)))){ headerRow = j; break; }
    }
    if(headerRow === -1) continue;

    /* rows run until a blank row or the next heading */
    let lastRow = headerRow;
    for(let j = headerRow + 1; j < rows.length; j++){
      const rc = tmNonEmpty(rows[j]);
      if(rc.length === 0) break;
      if(rc.length === 1 && /\b(20\d{2})\b/.test(String(rc[0]))
         && TM_MONTHS.some(mn => new RegExp("\\b"+mn+"\\b","i").test(String(rc[0])))) break;
      lastRow = j;
    }
    if(lastRow === headerRow) continue;

    blocks.push({
      month: mi + 1,
      year: yearM ? +yearM[1] : null,
      headerRow,
      firstRow: headerRow + 1,
      lastRow
    });
    i = lastRow;
  }
  return blocks;
}
```

```js
function tmCell(row, map, field){
  const i = map[field];
  if(i === undefined) return "";
  const v = tmRowCells(row)[i];
  return v==null ? "" : String(v).trim();
}
function tmRawCell(row, map, field){
  const i = map[field];
  if(i === undefined) return null;
  const v = tmRowCells(row)[i];
  return v==null ? null : v;
}
function tmPad(n){ return (n<10?"0":"") + n; }

/* Read every month block on every sheet into teamMbo records.
   sheets: [{ name, cells }] where cells is an array of row arrays.
   Returns { rows, months, flags } — flags are surfaced in the preview so
   nothing is repaired or dropped without the user seeing it. */
function tmReadSheets(sheets, ownerId, batchId, todayIso){
  const rows = [], flags = [], monthSet = {};

  (sheets||[]).forEach(sheet => {
    const isAction = /action\s*item/i.test(sheet.name || "");
    const kind = isAction ? "action" : "kr";
    const cells = sheet.cells || [];

    tmFindBlocks(cells, kind).forEach(block => {
      const map = tmColMap(tmRowCells(cells[block.headerRow]));
      if(map.title === undefined) return;

      /* an action heading may carry no year — take it from the first date
         we can actually read inside the block */
      let year = block.year;
      if(year === null){
        for(let r = block.firstRow; r <= block.lastRow; r++){
          const iso = tmParseDate(tmRawCell(cells[r], map, "deadline"));
          if(iso){ year = +iso.slice(0,4); break; }
        }
      }
      if(year === null){
        flags.push({ kind:"noyear", month:null, title:"(a block heading)",
                     detail:`"${String(tmNonEmpty(cells[block.headerRow-1])[0]||"")}" gives no year and no row in it has a readable date` });
        return;
      }

      const month = `${year}-${tmPad(block.month)}`;
      let lastObjective = "", lastObjNum = "";

      for(let r = block.firstRow; r <= block.lastRow; r++){
        const row = cells[r];
        const title = tmCell(row, map, "title");
        if(!title) continue;

        const objRaw = tmCell(row, map, "objective");
        if(objRaw) lastObjective = objRaw;
        const numRaw = tmCell(row, map, "objNum").replace(/[^0-9]/g, "");
        if(numRaw) lastObjNum = numRaw;

        const rawDeadline = tmRawCell(row, map, "deadline");
        const parsed = tmParseDate(rawDeadline);
        const rep = tmRepairDate(parsed, block.month);

        const priRaw = tmCell(row, map, "priority");
        const priNum = /^\d+(\.0+)?$/.test(priRaw) ? parseInt(priRaw, 10) : null;
        const band = priNum === null && priRaw
          ? (/^h/i.test(priRaw) ? "High" : /^(m|mid)/i.test(priRaw) ? "Medium" : /^l/i.test(priRaw) ? "Low" : "")
          : "";

        const rec = {
          id: genId("tmbo"),
          ownerId, month,
          type: kind,
          objNum: lastObjNum || "1",
          objective: kind === "action" ? "" : lastObjective,
          title,
          deadline: rep.iso,
          deadlineRaw: rawDeadline==null ? "" : String(rawDeadline).trim(),
          deadlineFixed: rep.fixed,
          rank: priNum,
          band,
          status: tmNormStatus(tmCell(row, map, "status")),
          timeEst: tmCell(row, map, "timeEst"),
          timeSpent: tmCell(row, map, "timeSpent"),
          comment: tmCell(row, map, "comment"),
          ref: tmCell(row, map, "ref"),
          importedAt: todayIso,
          importBatch: batchId
        };

        if(rep.fixed){
          flags.push({ kind:"repaired", month, title,
                       detail:`read as ${parsed}, corrected to ${rep.iso}` });
        } else if(rec.deadlineRaw && !rec.deadline){
          flags.push({ kind:"baddate", month, title,
                       detail:`could not read "${rec.deadlineRaw}"` });
        } else if(rec.deadline && Math.abs(+rec.deadline.slice(0,4) - year) > 5){
          flags.push({ kind:"wildyear", month, title,
                       detail:`year ${rec.deadline.slice(0,4)} is far from ${year} — check for a typo` });
        }

        monthSet[month] = true;
        rows.push(rec);
      }
    });
  });

  return { rows, months: Object.keys(monthSet).sort(), flags };
}

/* Their file holds every month, so the same file gets uploaded repeatedly.
   Drop this owner's rows for the months the file covers, then add the new
   ones — otherwise the list doubles on every upload. */
function tmReplaceMonths(existing, ownerId, months, incoming){
  const drop = new Set(months || []);
  const kept = (existing || []).filter(r =>
    !(r && r.ownerId === ownerId && drop.has(r.month)));
  return kept.concat(incoming || []);
}
```

```js
const TM_BUCKETS = [
  { key:"overdue", label:"⚠ Overdue" },
  { key:"week",    label:"This week" },
  { key:"soon",    label:"Next 3 weeks" },
  { key:"later",   label:"Later" },
  { key:"undated", label:"No deadline given" },
  { key:"done",    label:"Done" }
];

function tmDaysLate(rec, todayIso){
  if(!rec || !rec.deadline) return 0;
  return Math.round((new Date(todayIso+"T00:00:00Z") - new Date(rec.deadline+"T00:00:00Z")) / 86400000);
}

/* First match wins, and "done" is checked first so a completed item never
   shows up as overdue and an undated completed item lands in Done. */
function tmBucket(rec, todayIso){
  if(!rec) return "undated";
  if(rec.status === "Done") return "done";
  if(!rec.deadline) return "undated";
  const late = tmDaysLate(rec, todayIso);
  if(late > 0)   return "overdue";
  if(late >= -6) return "week";
  if(late >= -27) return "soon";
  return "later";
}
```
